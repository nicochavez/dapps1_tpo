-- ─── 14b) Nuevo dueño tras la adjudicación (productos.nuevo_duenio) ──────────
-- Guarda el ID DE PERSONA del ganador (o del admin, persona 1, si no hubo pujas). NO es FK a
-- 'duenios': el comprador no necesita ser dueño, se traza directo con personas. El consignante
-- original (productos.duenio = 20) se conserva para la vista "My Items".
ALTER TABLE public.productos ADD COLUMN IF NOT EXISTS nuevo_duenio bigint;
-- Si una corrida previa (mapeo viejo @ManyToOne DuenioEntity) dejó una FK a 'duenios' sobre esta
-- columna, se elimina: bloquearía guardar compradores que no son dueños (ej. persona 12).
DO $$
DECLARE r record;
BEGIN
  FOR r IN
    SELECT con.conname
    FROM pg_constraint con
    JOIN pg_attribute att ON att.attrelid = con.conrelid AND att.attnum = ANY (con.conkey)
    WHERE con.conrelid = 'public.productos'::regclass
      AND con.contype = 'f'
      AND att.attname = 'nuevo_duenio'
  LOOP
    EXECUTE 'ALTER TABLE public.productos DROP CONSTRAINT ' || quote_ident(r.conname);
  END LOOP;
END $$;
UPDATE public.productos SET nuevo_duenio = 13 WHERE identificador = 3;   -- Óleo Abstracto  -> oro (persona 13)
